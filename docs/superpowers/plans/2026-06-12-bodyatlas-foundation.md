# BodyAtlas Foundation Implementation Plan (Plan 1 of 4)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A deployed, signed-in, empty BodyAtlas: monorepo, D1 + Drizzle, Google OAuth + KV sessions, and a themed (3 themes), bilingual (EN/JA) React shell with 4-tab navigation and a quick-log FAB stub.

**Architecture:** One Cloudflare Worker serves the React SPA (static assets) and a Hono REST API under `/api/*` + `/auth/*`. D1 via Drizzle ORM with committed migrations; sessions in KV; shared Zod schemas in a `shared` workspace typed end-to-end via Hono RPC. Dev runs everything single-origin through `@cloudflare/vite-plugin`.

**Tech Stack:** TypeScript, Hono, Drizzle ORM, D1, KV, Zod (v4), React 19, Vite, `@cloudflare/vite-plugin`, TanStack Query, react-router, Vitest (`@cloudflare/vitest-pool-workers` for the worker, jsdom + Testing Library for the app), GitHub Actions.

**Spec:** `docs/superpowers/specs/2026-06-12-bodyatlas-design.md`. This plan covers the foundation only. Plan 2 = workout logging, Plan 3 = health tracking, Plan 4 = trends + Today dashboard.

**Conventions used throughout:**
- Run all commands from the repo root unless stated otherwise.
- Install dependencies with `@latest`-style bare installs (no pinned versions in this doc); the lockfile pins them.
- Commit after every task. Conventional commit messages.
- If a Cloudflare config shape rejects, check the current docs first: Vite plugin <https://developers.cloudflare.com/workers/vite-plugin/>, Vitest integration <https://developers.cloudflare.com/workers/testing/vitest-integration/>.

- [ ] **Step 3: Implement** — `worker/src/auth/session.ts`

```ts
export const SESSION_TTL_SECONDS = 60 * 60 * 24 * 30; // 30 days, rolling

const key = (token: string) => `session:${token}`;

export async function createSession(
  kv: KVNamespace,
  userId: string,
): Promise<string> {
  const token = crypto.randomUUID();
  await kv.put(key(token), userId, { expirationTtl: SESSION_TTL_SECONDS });
  return token;
}

export async function getSessionUserId(
  kv: KVNamespace,
  token: string,
): Promise<string | null> {
  const userId = await kv.get(key(token));
  if (userId !== null) {
    // Rolling expiry: refresh the TTL on every authenticated request.
    await kv.put(key(token), userId, { expirationTtl: SESSION_TTL_SECONDS });
  }
  return userId;
}

export async function deleteSession(
  kv: KVNamespace,
  token: string,
): Promise<void> {
  await kv.delete(key(token));
}
```

- [ ] **Step 4: Implement** — `worker/src/auth/middleware.ts` (exercised by tests in Task 7 via `/api/me`)

```ts
import { getCookie } from 'hono/cookie';
import { createMiddleware } from 'hono/factory';
import type { AppEnv } from '../types';
import { getSessionUserId } from './session';

export const SESSION_COOKIE = 'ba_session';

export const requireAuth = createMiddleware<AppEnv>(async (c, next) => {
  const token = getCookie(c, SESSION_COOKIE);
  const userId = token ? await getSessionUserId(c.env.SESSIONS, token) : null;
  if (!userId) {
    return c.json(
      { error: { code: 'unauthorized', message: 'Sign in required' } },
      401,
    );
  }
  c.set('userId', userId);
  await next();
});
```

```ts
import { Hono } from 'hono';
import { deleteCookie, getCookie, setCookie } from 'hono/cookie';
import { eq } from 'drizzle-orm';
import { getDb, schema } from '../db';
import type { AppEnv } from '../types';
import { SESSION_COOKIE } from './middleware';
import { createSession, deleteSession, SESSION_TTL_SECONDS } from './session';

const OAUTH_COOKIE = 'ba_oauth';
const GOOGLE_AUTH_URL = 'https://accounts.google.com/o/oauth2/v2/auth';
const GOOGLE_TOKEN_URL = 'https://oauth2.googleapis.com/token';

function base64UrlEncode(bytes: Uint8Array): string {
  return btoa(String.fromCharCode(...bytes))
    .replaceAll('+', '-')
    .replaceAll('/', '_')
    .replace(/=+$/, '');
}

function base64UrlDecode(input: string): string {
  return atob(input.replaceAll('-', '+').replaceAll('_', '/'));
}

async function sha256(input: string): Promise<Uint8Array> {
  const digest = await crypto.subtle.digest(
    'SHA-256',
    new TextEncoder().encode(input),
  );
  return new Uint8Array(digest);
}

function isHttps(requestUrl: string): boolean {
  return new URL(requestUrl).protocol === 'https:';
}

const badRequest = (message: string) =>
  ({ error: { code: 'oauth_invalid', message } }) as const;

export const authRoutes = new Hono<AppEnv>()
  .get('/google', async (c) => {
    const state = crypto.randomUUID();
    const verifier = base64UrlEncode(crypto.getRandomValues(new Uint8Array(32)));
    const challenge = base64UrlEncode(await sha256(verifier));
    const redirectUri = new URL('/auth/google/callback', c.req.url).toString();

    setCookie(c, OAUTH_COOKIE, JSON.stringify({ state, verifier }), {
      httpOnly: true,
      secure: isHttps(c.req.url),
      sameSite: 'Lax',
      path: '/auth',
      maxAge: 600,
    });

    const params = new URLSearchParams({
      client_id: c.env.GOOGLE_CLIENT_ID,
      redirect_uri: redirectUri,
      response_type: 'code',
      scope: 'openid email profile',
      state,
      code_challenge: challenge,
      code_challenge_method: 'S256',
    });
    return c.redirect(`${GOOGLE_AUTH_URL}?${params.toString()}`);
  })
  .get('/google/callback', async (c) => {
    const raw = getCookie(c, OAUTH_COOKIE);
    deleteCookie(c, OAUTH_COOKIE, { path: '/auth' });

    const code = c.req.query('code');
    const returnedState = c.req.query('state');
    if (!raw || !code || !returnedState) {
      return c.json(badRequest('Missing OAuth parameters'), 400);
    }

    let stored: { state: string; verifier: string };
    try {
      stored = JSON.parse(raw) as { state: string; verifier: string };
    } catch {
      return c.json(badRequest('Malformed OAuth cookie'), 400);
    }
    if (returnedState !== stored.state) {
      return c.json(badRequest('State mismatch'), 400);
    }

    const redirectUri = new URL('/auth/google/callback', c.req.url).toString();
    const tokenRes = await fetch(GOOGLE_TOKEN_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body: new URLSearchParams({
        client_id: c.env.GOOGLE_CLIENT_ID,
        client_secret: c.env.GOOGLE_CLIENT_SECRET,
        code,
        code_verifier: stored.verifier,
        grant_type: 'authorization_code',
        redirect_uri: redirectUri,
      }),
    });
    if (!tokenRes.ok) {
      return c.json(
        { error: { code: 'oauth_exchange_failed', message: 'Google sign-in failed' } },
        502,
      );
    }

    // The id_token arrives directly from Google's token endpoint over TLS,
    // so decoding without signature verification is safe here.
    const { id_token } = (await tokenRes.json()) as { id_token: string };
    const payload = id_token.split('.')[1];
    if (!payload) return c.json(badRequest('Malformed id_token'), 400);
    const claims = JSON.parse(base64UrlDecode(payload)) as {
      sub: string;
      email: string;
      name?: string;
    };

    const db = getDb(c.env.DB);
    const existing = await db.query.users.findFirst({
      where: eq(schema.users.googleSub, claims.sub),
    });
    const userId = existing?.id ?? crypto.randomUUID();
    const displayName = claims.name ?? claims.email;
    if (existing) {
      await db
        .update(schema.users)
        .set({ email: claims.email, name: displayName })
        .where(eq(schema.users.id, userId));
    } else {
      await db.insert(schema.users).values({
        id: userId,
        googleSub: claims.sub,
        email: claims.email,
        name: displayName,
      });
    }

    const token = await createSession(c.env.SESSIONS, userId);
    setCookie(c, SESSION_COOKIE, token, {
      httpOnly: true,
      secure: isHttps(c.req.url),
      sameSite: 'Lax',
      path: '/',
      maxAge: SESSION_TTL_SECONDS,
    });
    return c.redirect('/');
  })
  .post('/logout', async (c) => {
    const token = getCookie(c, SESSION_COOKIE);
    if (token) await deleteSession(c.env.SESSIONS, token);
    deleteCookie(c, SESSION_COOKIE, { path: '/' });
    return c.redirect('/');
  });
```

import type { Locale } from '@bodyatlas/shared';

const catalogs: Record<Locale, Record<MessageKey, string>> = { en, ja };

const I18nContext = createContext<Locale>('en');

export function I18nProvider({
  locale,
  children,
}: {
  locale: Locale;
  children: ReactNode;
}) {
  return <I18nContext.Provider value={locale}>{children}</I18nContext.Provider>;
}

export function useI18n() {
  const locale = useContext(I18nContext);
  const t = (key: MessageKey): string => catalogs[locale][key];
  return { locale, t };
}

export function detectBrowserLocale(): Locale {
  return navigator.language.toLowerCase().startsWith('ja') ? 'ja' : 'en';
}

import type { Theme } from '@bodyatlas/shared';

const THEME_STORAGE_KEY = 'ba_theme';

export function applyTheme(theme: Theme) {
  document.documentElement.dataset.theme = theme;
  try {
    localStorage.setItem(THEME_STORAGE_KEY, theme);
  } catch {
    // Storage may be unavailable (private mode); the attribute alone is fine.
  }
}

export function storedTheme(): Theme | null {
  try {
    return localStorage.getItem(THEME_STORAGE_KEY) as Theme | null;
  } catch {
    return null;
  }
}

/** Applies `theme` to <html data-theme> whenever it changes. */
export function ThemeProvider({
  theme,
  children,
}: {
  theme: Theme;
  children: ReactNode;
}) {
  useEffect(() => {
    applyTheme(theme);
  }, [theme]);
  return <>{children}</>;
}

import { useMutation, useQueryClient } from '@tanstack/react-query';
import { LOCALES, THEMES, type UpdateMe } from '@bodyatlas/shared';
import { useI18n, type MessageKey } from '../i18n';
import { api } from '../lib/api';
import { useMe } from '../lib/useMe';

export function SettingsPage() {
  const { t } = useI18n();
  const { data: me } = useMe();
  const queryClient = useQueryClient();

  const update = useMutation({
    mutationFn: async (patch: UpdateMe) => {
      const res = await api.me.$patch({ json: patch });
      if (!res.ok) throw new Error(`PATCH /api/me failed: ${res.status}`);
      return res.json();
    },
    onSuccess: (updated) => {
      queryClient.setQueryData(['me'], updated);
    },
  });

  if (!me) return null;

  const fieldStyle = {
    display: 'block',
    width: '100%',
    maxWidth: 320,
    padding: 10,
    marginTop: 6,
    background: 'var(--surface-2)',
    color: 'var(--text)',
    border: '1px solid var(--border)',
    borderRadius: 'var(--radius)',
  } as const;

  return (
    <section>
      <h1>{t('settings.title')}</h1>

      <label style={{ display: 'block', marginBottom: 16 }}>
        {t('settings.theme')}
        <select
          aria-label={t('settings.theme')}
          style={fieldStyle}
          value={me.theme}
          onChange={(e) => update.mutate({ theme: e.target.value as UpdateMe['theme'] })}
        >
          {THEMES.map((theme) => (
            <option key={theme} value={theme}>
              {t(`settings.theme.${theme}` as MessageKey)}
            </option>
          ))}
        </select>
      </label>

      <label style={{ display: 'block', marginBottom: 16 }}>
        {t('settings.language')}
        <select
          aria-label={t('settings.language')}
          style={fieldStyle}
          value={me.locale}
          onChange={(e) => update.mutate({ locale: e.target.value as UpdateMe['locale'] })}
        >
          {LOCALES.map((locale) => (
            <option key={locale} value={locale}>
              {t(`settings.language.${locale}` as MessageKey)}
            </option>
          ))}
        </select>
      </label>

      {update.isSuccess && (
        <p style={{ color: 'var(--positive)' }}>{t('settings.saved')}</p>
      )}
      {update.isError && (
        <p style={{ color: 'var(--negative)' }}>{t('error.generic')}</p>
      )}

      <form method="post" action="/auth/logout">
        <button
          type="submit"
          style={{
            marginTop: 24,
            padding: '10px 20px',
            background: 'var(--surface-2)',
            color: 'var(--text)',

            borderRadius: 'var(--radius)',
